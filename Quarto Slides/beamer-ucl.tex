% Frame title with a short violet rule underneath

% Footer: lecture name (set \uclfooter in the .qmd), frame number and the UCL logo

% Title page on a full-bleed UCL ink background
\setbeamertemplate{title page}{%
  \begin{tikzpicture}[remember picture,overlay]
    \fill[uclink] (current page.south west) rectangle (current page.north east);
    \node[anchor=north east, inner sep=0pt] at ([xshift=-0.8cm,yshift=-0.7cm]current page.north east)
      {\includegraphics[height=1.5cm]{figures/ucl_logo.pdf}};
  \end{tikzpicture}%
  \vfill
  {\color{uclviolet}\rule{2.4em}{3pt}}\par\vspace{0.6em}
  {\color{white}\usebeamerfont{title}\bfseries\inserttitle\par}
  \vspace{0.3em}{\color{uclviolet!40!white}\usebeamerfont{subtitle}\insertsubtitle\par}
  \vspace{1.6em}
  {\color{white}\bfseries\insertauthor\par}
  \vspace{0.2em}
  {\color{uclviolet!40!white}\small\insertinstitute\par}
  \vfill
}
